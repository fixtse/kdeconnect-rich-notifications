cmake_minimum_required(VERSION 3.22)
project(kdeconnect-rich-notifications VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This plugin currently supports Linux only")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
include(CTest)
include(GNUInstallDirs)
find_package(ECM 6.0 REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(ECMQtDeclareLoggingCategory)
find_package(Qt6 6.7 REQUIRED COMPONENTS Core Gui Network DBus Test)
find_package(KF6CoreAddons REQUIRED)
find_package(KF6I18n REQUIRED)
find_package(KF6IconThemes REQUIRED)
find_package(KF6Config REQUIRED)
find_package(KF6Service REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DBus REQUIRED IMPORTED_TARGET dbus-1)
set(KDECONNECT_SOURCE_DIR "" CACHE PATH "Unmodified KDE Connect source matching the installed version")
if(NOT EXISTS "${KDECONNECT_SOURCE_DIR}/core/kdeconnectplugin.h")
    message(FATAL_ERROR "Set KDECONNECT_SOURCE_DIR to matching KDE Connect source; see README.md")
endif()
file(READ "${KDECONNECT_SOURCE_DIR}/CMakeLists.txt" upstream_version)
foreach(part MAJOR MINOR MICRO)
    string(REGEX MATCH "RELEASE_SERVICE_VERSION_${part} \"([0-9]+)\"" version_match "${upstream_version}")
    if(NOT version_match)
        message(FATAL_ERROR "Cannot determine KDE Connect version")
    endif()
    set(KDE_VERSION_${part} "${CMAKE_MATCH_1}")
endforeach()
set(KDECONNECT_VERSION "${KDE_VERSION_MAJOR}.${KDE_VERSION_MINOR}.${KDE_VERSION_MICRO}")
find_library(KDECONNECT_CORE_LIBRARY NAMES kdeconnectcore libkdeconnectcore.so.${KDE_VERSION_MAJOR} REQUIRED)
add_library(kdeconnectcore SHARED IMPORTED)
set_target_properties(kdeconnectcore PROPERTIES IMPORTED_LOCATION "${KDECONNECT_CORE_LIBRARY}")
include(GenerateExportHeader)
generate_export_header(kdeconnectcore EXPORT_FILE_NAME "${CMAKE_CURRENT_BINARY_DIR}/kdeconnectcore_export.h")
add_compile_definitions(QT_NO_KEYWORDS QT_NO_CAST_FROM_ASCII QT_NO_CAST_TO_ASCII)
include_directories(src "${CMAKE_CURRENT_BINARY_DIR}" "${KDECONNECT_SOURCE_DIR}" "${KDECONNECT_SOURCE_DIR}/core")
set(shared_sources src/notificationslistener.cpp src/notifyingapplication.cpp
    src/notificationimagehint.cpp src/pw_libnotificationmanager/notification.cpp)
set(shared_libraries kdeconnectcore Qt6::Core Qt6::Gui Qt6::Network Qt6::DBus
    KF6::CoreAddons KF6::I18n KF6::IconThemes KF6::ConfigCore KF6::Service PkgConfig::DBus)
add_library(kdeconnect_richnotifications MODULE ${shared_sources}
    src/sendnotificationsplugin.cpp src/dbusnotificationslistener.cpp)
set_target_properties(kdeconnect_richnotifications PROPERTIES PREFIX "")
ecm_qt_declare_logging_category(kdeconnect_richnotifications
    HEADER plugin_sendnotifications_debug.h IDENTIFIER KDECONNECT_PLUGIN_SENDNOTIFICATIONS
    CATEGORY_NAME kdeconnect.plugin.richnotifications DEFAULT_SEVERITY Warning)
target_link_libraries(kdeconnect_richnotifications PRIVATE ${shared_libraries})
if(BUILD_TESTING)
    add_executable(notificationimagetest src/notificationimagetest.cpp ${shared_sources})
    target_link_libraries(notificationimagetest PRIVATE ${shared_libraries} Qt6::Test)
    add_test(NAME notificationimagetest COMMAND notificationimagetest)
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/compatibility.json"
    "{\"kdeconnect\":\"${KDECONNECT_VERSION}\",\"qt\":\"${Qt6Core_VERSION}\",\"project\":\"${PROJECT_VERSION}\"}\n")
install(TARGETS kdeconnect_richnotifications LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}/qt6/plugins/kdeconnect")
install(FILES src/kdeconnect_richnotifications_config.qml DESTINATION share/kdeconnect)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/compatibility.json" DESTINATION share/kdeconnect-rich-notifications)
